cmake_minimum_required(VERSION 3.24)
project(PaleoDataService VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(PDS_BUILD_DEMO "Build the desktop demonstration" ${PROJECT_IS_TOP_LEVEL})
option(PDS_BUILD_TOOLS "Build command-line tools" ${PROJECT_IS_TOP_LEVEL})
option(PDS_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(PDS_BUILD_TESTS "Build the service tests" ${PROJECT_IS_TOP_LEVEL})
option(PDS_INSTALL "Install the standalone SDK" ${PROJECT_IS_TOP_LEVEL})
set(PDS_SEISMIC_ENGINE_TARGET "" CACHE STRING "Existing compatible seismic engine target (optional)")
set(PDS_SEISMIC_INCLUDE_DIR "" CACHE PATH "Headers matching the existing seismic engine")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Sql)
if(PDS_BUILD_DEMO OR PDS_BUILD_TESTS)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
endif()
find_package(ZLIB REQUIRED)
if(PDS_SEISMIC_ENGINE_TARGET)
  if(NOT TARGET "${PDS_SEISMIC_ENGINE_TARGET}" OR NOT IS_DIRECTORY "${PDS_SEISMIC_INCLUDE_DIR}")
    message(FATAL_ERROR "An existing seismic target requires its matching include directory")
  endif()
  set(_pds_engine "${PDS_SEISMIC_ENGINE_TARGET}")
else()
  include(cmake/SeismicEngine.cmake)
  set(_pds_engine pds_sbm)
endif()
file(GLOB PARSER_SOURCES CONFIGURE_DEPENDS src/parsers/*.cpp)
file(GLOB STORE_SOURCES CONFIGURE_DEPENDS src/store/*.cpp)
file(GLOB FLOW_SOURCES CONFIGURE_DEPENDS src/flow/*.cpp)
file(GLOB SERVICE_SOURCES CONFIGURE_DEPENDS src/services/*.cpp src/adapters/*.cpp)
add_library(paleo_data_service STATIC ${PARSER_SOURCES} ${STORE_SOURCES} ${FLOW_SOURCES} ${SERVICE_SOURCES} vendor/paleo_io/ziparchive.cpp)
target_include_directories(paleo_data_service PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src vendor/paleo_io vendor/sbm/src vendor/glm vendor/segyio/include)
target_compile_definitions(paleo_data_service PRIVATE PALEO_HAVE_ZLIB)
target_link_libraries(paleo_data_service PRIVATE Qt6::Core Qt6::Gui Qt6::Sql ZLIB::ZLIB ${_pds_engine})
if(WIN32)
  target_link_libraries(paleo_data_service PRIVATE bcrypt)
endif()
add_library(Paleo::paleo_data_service ALIAS paleo_data_service)
if(PDS_SEISMIC_ENGINE_TARGET)
  target_include_directories(paleo_data_service BEFORE PRIVATE "${PDS_SEISMIC_INCLUDE_DIR}")
endif()
if(MSVC)
  target_compile_options(paleo_data_service PRIVATE /utf-8 /W4)
  add_compile_options(/utf-8)
endif()
file(GLOB DEMO_SOURCES CONFIGURE_DEPENDS demo/*.cpp demo/*.h)
if(PDS_BUILD_DEMO AND DEMO_SOURCES)
  add_executable(paleo_data_demo WIN32 ${DEMO_SOURCES})
  target_link_libraries(paleo_data_demo PRIVATE paleo_data_service Qt6::Widgets)
endif()
if(PDS_BUILD_TOOLS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/service_cli.cpp")
  add_executable(paleo_data_cli tools/service_cli.cpp)
  target_link_libraries(paleo_data_cli PRIVATE paleo_data_service Qt6::Core)
endif()
if(PDS_BUILD_TOOLS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/seismic_acceptance.cpp")
  add_executable(seismic_acceptance tools/seismic_acceptance.cpp)
  target_link_libraries(seismic_acceptance PRIVATE paleo_data_service Qt6::Core)
endif()
if(PDS_BUILD_TOOLS)
  add_executable(segyio_acceptance tools/segyio_acceptance.cpp)
  target_link_libraries(segyio_acceptance PRIVATE paleo_data_service Qt6::Core)
  add_executable(timeslice_benchmark tools/timeslice_benchmark.cpp)
  target_link_libraries(timeslice_benchmark PRIVATE paleo_data_service Qt6::Core)
endif()
if(PDS_BUILD_TESTS)
include(CTest)
if(NOT PDS_SEISMIC_ENGINE_TARGET)
  add_executable(segyio_bounds_tests tests/segyio_bounds_tests.cpp)
  target_include_directories(segyio_bounds_tests PRIVATE vendor/segyio/include)
  target_link_libraries(segyio_bounds_tests PRIVATE pds_segyio)
  add_test(NAME segyio_bounds_tests COMMAND segyio_bounds_tests)
endif()
foreach(test_name IN ITEMS well_plot_tests aux_parser_tests text_parser_tests seismic_service_tests seismic_hd_tests timeslice_batch_tests spreadsheet_xml_tests core_asset_tests format_compatibility_tests segyio_access_tests seismic_processing_tests import_performance_tests file_digest_tests stratigraphic_framework_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_name}.cpp")
    add_executable(${test_name} tests/${test_name}.cpp)
    target_include_directories(${test_name} PRIVATE src)
    target_link_libraries(${test_name} PRIVATE paleo_data_service Qt6::Core Qt6::Sql)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_tests.cpp")
  add_executable(service_tests tests/service_tests.cpp)
  target_link_libraries(service_tests PRIVATE paleo_data_service Qt6::Core)
  add_test(NAME service_contract COMMAND service_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/file_graph_tests.cpp")
  add_executable(file_graph_tests tests/file_graph_tests.cpp)
  target_link_libraries(file_graph_tests PRIVATE paleo_data_service Qt6::Core Qt6::Sql)
  add_test(NAME file_graph_contract COMMAND file_graph_tests)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/coordinate_system_tests.cpp")
  add_executable(coordinate_system_tests tests/coordinate_system_tests.cpp)
  target_link_libraries(coordinate_system_tests PRIVATE paleo_data_service Qt6::Core Qt6::Sql)
  add_test(NAME coordinate_system_contract COMMAND coordinate_system_tests)
endif()
endif() # PDS_BUILD_TESTS

file(GLOB STRATIGRAPHY_CORE_SOURCES CONFIGURE_DEPENDS src/stratigraphy/core/*.cpp)
add_library(paleo_stratigraphy_core STATIC ${STRATIGRAPHY_CORE_SOURCES})
target_include_directories(paleo_stratigraphy_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(paleo_stratigraphy_core PRIVATE vendor/delaunator)
set_target_properties(paleo_stratigraphy_core PROPERTIES AUTOMOC OFF)
add_library(paleo_stratigraphy_service STATIC src/stratigraphy/service/workflow.cpp src/stratigraphy/service/quick_workflow.cpp src/stratigraphy/service/conventions.cpp src/stratigraphy/service/time_depth_selection.cpp)
target_include_directories(paleo_stratigraphy_service PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(paleo_stratigraphy_service PUBLIC paleo_stratigraphy_core paleo_data_service PRIVATE Qt6::Core)
set_target_properties(paleo_stratigraphy_service PROPERTIES AUTOMOC OFF)
add_library(Paleo::paleo_stratigraphy_core ALIAS paleo_stratigraphy_core)
add_library(Paleo::paleo_stratigraphy_service ALIAS paleo_stratigraphy_service)

if(MSVC)
 target_compile_options(paleo_stratigraphy_core PRIVATE /utf-8)
 target_compile_options(paleo_stratigraphy_service PRIVATE /utf-8)
endif()

if(PDS_INSTALL)
set(_pds_install_targets paleo_data_service paleo_stratigraphy_core paleo_stratigraphy_service)
if(NOT PDS_SEISMIC_ENGINE_TARGET)
  list(APPEND _pds_install_targets pds_sbm pds_segyio)
else()
  message(FATAL_ERROR "PDS_INSTALL requires the standalone seismic engine")
endif()
install(TARGETS ${_pds_install_targets} EXPORT PaleoDataServiceTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT PaleoDataServiceTargets FILE PaleoDataServiceTargets.cmake NAMESPACE Paleo:: DESTINATION lib/cmake/PaleoDataService)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/PaleoDataServiceConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfig.cmake" INSTALL_DESTINATION lib/cmake/PaleoDataService)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfigVersion.cmake" DESTINATION lib/cmake/PaleoDataService)
endif()
